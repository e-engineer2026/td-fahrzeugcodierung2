/**
 * TD Fahrzeugcodierung - PayPal.Me Gmail-Import
 * Wird im eigenen Google-Konto unter https://script.google.com angelegt.
 * Es werden NUR Zahlungseingangs-E-Mails von PayPal an die eigene TD-Website gesendet.
 * Eine E-Mail schaltet NIEMALS eine PDF frei.
 *
 * Einrichtung:
 * 1. Einmaligen Schluessel in https://td-fahrzeugcodierung.de/konto/pdf-verwaltung erzeugen.
 * 2. In Google Apps Script > Projekteinstellungen > Skripteigenschaften:
 *      TD_PDF_WEBHOOK_KEY = <einmaliger Schluessel aus der PDF-Verwaltung>
 * 3. Diese Datei als Code.gs einsetzen und tdInstallTrigger einmal ausfuehren.
 * 4. Gmail-Berechtigung bestaetigen. Der Trigger laeuft danach alle 15 Minuten.
 */

const TD_PDF_ENDPOINT = 'https://td-fahrzeugcodierung.de/api/guide-shop/gmail-receipts';
const TD_PDF_QUERY = '(from:paypal.com OR from:paypal.de) newer_than:7d -in:sent -in:spam -in:trash';
const TD_PDF_RECEIPT = /(zahlung erhalten|zahlung von|hat ihnen .{0,60}gesendet|geld erhalten|received a payment|payment received|you(?:'|’)ve received|sent you money|has sent you money)/i;

function tdInstallTrigger() {
  const existing = ScriptApp.getProjectTriggers().filter(
    trigger => trigger.getHandlerFunction() === 'tdForwardPayPalMessages');
  existing.forEach(trigger => ScriptApp.deleteTrigger(trigger));
  ScriptApp.newTrigger('tdForwardPayPalMessages').timeBased().everyMinutes(15).create();
  tdForwardPayPalMessages();
}

function tdForwardPayPalMessages() {
  const props = PropertiesService.getScriptProperties();
  const key = props.getProperty('TD_PDF_WEBHOOK_KEY');
  if (!key || !/^[A-Za-z0-9_-]{43}$/.test(key)) {
    throw new Error('TD_PDF_WEBHOOK_KEY fehlt oder ist ungueltig.');
  }
  const sent = JSON.parse(props.getProperty('TD_PDF_SENT_IDS') || '[]');
  const sentSet = new Set(sent);
  const candidates = [];
  const threads = GmailApp.search(TD_PDF_QUERY, 0, 30);
  for (const thread of threads) {
    for (const message of thread.getMessages()) {
      const id = message.getId();
      if (sentSet.has(id)) continue;
      const from = message.getFrom();
      const address = (from.match(/<([^<>]+)>/) || [null, from])[1].trim().toLowerCase();
      const domain = address.split('@')[1] || '';
      if (!/^(?:[a-z0-9-]+\.)*paypal\.(?:com|de)$/.test(domain)) continue;
      const subject = message.getSubject() || '';
      if (!TD_PDF_RECEIPT.test(subject)) continue;
      if (message.getDate().getTime() < Date.now() - 14 * 86400000) continue;
      candidates.push({
        messageId: id,
        sender: from.slice(0, 300),
        subject: subject.slice(0, 300),
        messageAt: message.getDate().toISOString(), // Mail-Zeit, NICHT PayPal-Zahlungszeit
        plainBody: (message.getPlainBody() || '').slice(0, 5000)
      });
    }
  }
  candidates.sort((a, b) => a.messageAt.localeCompare(b.messageAt));
  const batch = candidates.slice(0, 12);
  if (batch.length === 0) return;
  const response = UrlFetchApp.fetch(TD_PDF_ENDPOINT, {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + key },
    payload: JSON.stringify({ messages: batch }),
    muteHttpExceptions: true,
  });
  const code = response.getResponseCode();
  if (code < 200 || code >= 300) {
    throw new Error('TD Gmail-Import fehlgeschlagen: HTTP ' + code);
  }
  const updatedIds = sent.concat(batch.map(item => item.messageId)).slice(-250);
  props.setProperty('TD_PDF_SENT_IDS', JSON.stringify(updatedIds));
  console.log('TD PDF: ' + batch.length + ' PayPal-E-Mails an die Website uebertragen. Keine Freigabe.');
}
